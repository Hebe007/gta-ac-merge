cmake_minimum_required(VERSION 3.20)
project(gta_ac_merge CXX)
# Static MSVC runtime: the .asi needs no Visual C++ redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(linkcommon common/link_math.cpp)
target_include_directories(linkcommon PUBLIC common third_party)
add_library(gtalogic gta/src/link_client.cpp gta/src/puppet.cpp)
target_include_directories(gtalogic PUBLIC gta/src)
target_link_libraries(gtalogic PUBLIC linkcommon)
# ScriptHookV SDK: unzip it into Scripthook/ (any sub-folder layout). Not committed (Rockstar/Alexander Blade licence).
if(WIN32)
  file(GLOB_RECURSE SHV_CALLER "${CMAKE_SOURCE_DIR}/Scripthook/*nativeCaller.h")
  file(GLOB_RECURSE SHV_LIB "${CMAKE_SOURCE_DIR}/Scripthook/*ScriptHookV.lib")
  if(SHV_CALLER AND SHV_LIB)
    list(GET SHV_CALLER 0 SHV_CALLER)
    list(GET SHV_LIB 0 SHV_LIB)
    get_filename_component(SHV_INC "${SHV_CALLER}" DIRECTORY)
    message(STATUS "ScriptHookV SDK: ${SHV_INC}")
    add_subdirectory(gta)
  else()
    message(WARNING "ScriptHookV SDK not found under Scripthook/ - the GTA .asi will not be built")
  endif()
endif()
if(NOT WIN32 OR BUILD_TESTING)
  enable_testing()
  add_subdirectory(tests)
endif()
